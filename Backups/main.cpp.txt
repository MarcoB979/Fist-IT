// OSSM STROKE ENGINE COMMUNICATION
bool SendOSSMCommand(int command, float value) {
  if (!OSSM_paired) {
    LogDebug("OSSM not paired - attempting to connect");
    
    // Try to pair with OSSM by sending heartbeat
    struct_message ossmMessage;
    memset(&ossmMessage, 0, sizeof(ossmMessage));
    ossmMessage.esp_command = HEARTBEAT;
    ossmMessage.esp_heartbeat = true;
    ossmMessage.esp_target = OSSM_ID;
    ossmMessage.esp_sender = FIST_ID;
    
    esp_err_t result = esp_now_send(OSSM_Address, (uint8_t *) &ossmMessage, sizeof(ossmMessage));
    if (result == ESP_OK) {
      LogDebug("OSSM heartbeat sent for pairing");
    }
    return false;
  }
  
  // Send command to paired OSSM
  struct_message ossmMessage;
  memset(&ossmMessage, 0, sizeof(ossmMessage));
  ossmMessage.esp_connected = true;
  ossmMessage.esp_command = command;
  ossmMessage.esp_value = value;
  ossmMessage.esp_target = OSSM_ID;
  ossmMessage.esp_sender = FIST_ID;
  
  esp_err_t result = esp_now_send(OSSM_Address, (uint8_t *) &ossmMessage, sizeof(ossmMessage));
  if (result == ESP_OK) {
    LogDebugFormatted("OSSM command sent: CMD=%d, VALUE=%.2f, MAC ADDRESS: %02X:%02X:%02X:%02X:%02X:%02X\n", command, value, OSSM_Address[0], OSSM_Address[1], OSSM_Address[2], OSSM_Address[3], OSSM_Address[4], OSSM_Address[5] );
    return true;
  } else {
    LogDebugFormatted("Failed to send OSSM command: CMD=%d\n", command);
    return false;
  }
}


// Callback when data is sent
void OnDataSent(const uint8_t *mac_addr, esp_now_send_status_t status) {
  bool isBroadcast = false;
  if (mac_addr != nullptr) {
    isBroadcast = (mac_addr[0] == 0xFF && mac_addr[1] == 0xFF && mac_addr[2] == 0xFF &&
                   mac_addr[3] == 0xFF && mac_addr[4] == 0xFF && mac_addr[5] == 0xFF);
  }

  LogDebugFormattedPRIO(
    "ESP-NOW TX: %s to %02X:%02X:%02X:%02X:%02X:%02X (%s) cmd=%d target=%d sender=%d hb=%d conn=%d\n",
    status == ESP_NOW_SEND_SUCCESS ? "OK" : "FAIL",
    mac_addr ? mac_addr[0] : 0,
    mac_addr ? mac_addr[1] : 0,
    mac_addr ? mac_addr[2] : 0,
    mac_addr ? mac_addr[3] : 0,
    mac_addr ? mac_addr[4] : 0,
    mac_addr ? mac_addr[5] : 0,
    isBroadcast ? "BROADCAST" : "UNICAST",
    outgoingcontrol.esp_command,
    outgoingcontrol.esp_target,
    outgoingcontrol.esp_sender,
    outgoingcontrol.esp_heartbeat ? 1 : 0,
    outgoingcontrol.esp_connected ? 1 : 0
  );

  if (status ==0){
    success = "Delivery Success :)";
  }
  else{
    success = "Delivery Fail :(";
  }
}

void OnDataRecvGYRO(const uint8_t * mac, const uint8_t *incomingData, int len) {
  bool needToAddPeer = false;
//  LogDebugPRIO("Time received: " + String(millis()));

  memcpy(&incomingcontrol, incomingData, sizeof(incomingcontrol));
  if (incomingcontrol.esp_sender == FIST_IT_GYRO_ID && !FIST_IT_RC_paired) {
    // Fist-IT RC trying to connect for first time
    esp_now_del_peer(peerInfo.peer_addr); // Clear any existing peer
    memcpy(FIST_IT_RC_Address, mac, 6);
    memcpy(peerInfo.peer_addr, mac, 6);
    needToAddPeer = true;
    LogDebugPRIO("Setting up Fist-IT RC peer");
    LogDebugFormattedPRIO("Gyro MAC received: %02X:%02X:%02X:%02X:%02X:%02X\n", 
                      mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]);
  }

  if (incomingcontrol.esp_sender == FIST_IT_GYRO_ID && FIST_IT_RC_paired) {
    // Fist-IT RC is already paired, just switch to its peer address
    memcpy(peerInfo.peer_addr, FIST_IT_RC_Address, 6);
    //LogDebugPRIO("Switching to Fist-IT RC peer");
  }

      if (needToAddPeer) {
      peerInfo.channel = 1;
      peerInfo.encrypt = false;
      
      esp_err_t result = esp_now_add_peer(&peerInfo);
      if (result == ESP_OK) {
        LogDebugFormattedPRIO("Peer added: %02X:%02X:%02X:%02X:%02X:%02X from ID %d\n", 
                          mac[0], mac[1], mac[2], mac[3], mac[4], mac[5], incomingcontrol.esp_sender);
        
      if (incomingcontrol.esp_sender == FIST_IT_GYRO_ID) {
          FIST_IT_RC_paired = true;
          LogDebugPRIO("Fist-IT RC Connected");
        }
      }
      else {
        LogDebugPRIO("Failed to add peer");
      }
    }

        // ALWAYS ALLOW SAFETY STATUS UPDATES - they bypass all safety checks
    if (incomingcontrol.esp_sender == FIST_IT_GYRO_ID && incomingcontrol.esp_command == FSTOP) {
      gyroSafetyEngaged = (incomingcontrol.esp_value > 0.5f); // 1=SAFE, 0=ACTIVE
      LogDebugFormatted("SAFETY: Gyro safety %s\n", gyroSafetyEngaged ? "ENGAGED" : "RELEASED");
      
      // Send acknowledgment back to gyro to confirm connection
      outgoingcontrol.esp_command = FSTOP;
      outgoingcontrol.esp_value = gyroSafetyEngaged ? 1.0f : 0.0f;
      outgoingcontrol.esp_target = FIST_IT_GYRO_ID;
      outgoingcontrol.esp_sender = FIST_ID;
      esp_now_send(mac, (uint8_t *) &outgoingcontrol, sizeof(outgoingcontrol));
      return; // Don't process as movement command
    }
    
    bool allowMovementCommand = true;
    unsigned long currentTime = millis();

    if (incomingcontrol.esp_sender == FIST_IT_GYRO_ID) {
      
      // Check if gyro movement should be blocked by safety system
      bool m5RecentlyActive = (currentTime - lastM5CommandTime) < DEVICE_TIMEOUT;
      
      if (gyroSafetyEngaged) {
        LogDebugPRIO("SAFETY: Gyro movement blocked - Safety is engaged");
        allowMovementCommand = false;
      }
      else if (m5RecentlyActive) {
        LogDebugPRIO("SAFETY: Gyro movement blocked - M5 Remote has priority");
        activeController = M5_ID;
        allowMovementCommand = false;
      }
      else {
        // Allow gyro control when safety is released AND M5 is inactive
        activeController = FIST_IT_GYRO_ID;
        LogDebugFormatted("SAFETY: Gyro movement allowed - M5 inactive and safety released, command: %d\n", incomingcontrol.esp_command);
      }
    }

    if (!allowMovementCommand) {
      LogDebugPRIO("Movement command blocked by safety system");
      return;
    }
          //HANDLE ACTUAL COMMANDS SENT FROM M5 REMOTE
    
    if (incomingcontrol.esp_sender == FIST_IT_GYRO_ID) {

      // M5 Remote always has movement priority
      activeController = FIST_IT_GYRO_ID;
      LogDebug("SAFETY: Gyro Remote active - has priority control");
      
      // Check if M5 is sending a stop command (zero velocity)
      bool isStopCommand = (abs(incomingcontrol.esp_value) < 0.01f);
      if (isStopCommand) {
        // Stop command - clear M5 activity to allow gyro control later
        activeController = 0;
        LogDebug("SAFETY: Gyro Remote STOP - clearing priority for gyro connection");
      }
    } 
    // --- Minimal Command Execution ---
    switch(incomingcontrol.esp_command)
    {

      case FANGLE: {
        // Only print debug for new angle/speed from gyro above
        // Map -180 to +180 degrees directly to steps
        float angleValue = incomingcontrol.esp_value;
        if (DirectionInvert) {
            angleValue = -angleValue;
        }
        EndPosition = (int)((stepsPerRotation/360.0) * (angleValue * rotationScale));
        MotorStatus = ANGLE_MODE;
        speedValue =  incomingcontrol.esp_speed;;
        LogDebugPRIO("Angle received: " + String(incomingcontrol.esp_value) + ", Mapped Steps=" + String(EndPosition) + ", Speed=" + String(speedValue));
        //currentRotationVelocity = incomingcontrol.esp_speed;
//        LogDebugFormattedPRIO("Current Milis: %lu, FROTATION received: Angle=%.2f, Mapped Steps=%d, RotationScale=%.2f, Velocity=%.2f\n", millis(), incomingcontrol.esp_value, EndPosition, rotationScale, currentRotationVelocity);
        break;
      }
      case FPAUSE: {
        F_Pause = incomingcontrol.esp_value;
        break;
      }
      case FROTATION_SCALE: {
        rotationScale = incomingcontrol.esp_value;
        break;
      }
      case FRESPONSIVENESS: {
        responsivenessLevel = (int)incomingcontrol.esp_value;
        break;
      }
      case FVELOCITY: {
        // Only print debug for new angle/speed from gyro above
        currentRotationVelocity = incomingcontrol.esp_value;
        break;
      }
      case ON: {
        MotorStatus=START;
        break;
      }
      case OFF: {
        MotorStatus=STOP;
        break;
      }
      case CONNECT:
      case HEARTBEAT: {
        outgoingcontrol.esp_target = FIST_IT_GYRO_ID;
        outgoingcontrol.esp_sender = FIST_ID;
        outgoingcontrol.esp_command = HEARTBEAT;
        esp_err_t result = esp_now_send(mac, (uint8_t *) &outgoingcontrol, sizeof(outgoingcontrol));
        if (result == ESP_OK) {
          FIST_IT_RC_paired = true;
          LogDebugPRIO("Fist-IT RC Connected via HEARTBEAT");}
        }
        break;
    }
    
    
    // Send appropriate response directly to the sender (no peer switching needed for responses)
    outgoingcontrol.esp_sender = FIST_ID;
    outgoingcontrol.esp_command = CONNECT;
    outgoingcontrol.esp_value = 99;
    outgoingcontrol.esp_target = FIST_IT_GYRO_ID;
    // Send response directly to sender's MAC address
    esp_err_t sendResult = esp_now_send(mac, (uint8_t *) &outgoingcontrol, sizeof(outgoingcontrol));

} 

// Callback when data is received
void OnDataRecv(const uint8_t * mac, const uint8_t *incomingData, int len) {

    // --- Angle jump protection ---
    const float MAX_ANGLE_JUMP_DEGREES = 30.0f; // Maximum allowed jump in degrees
    const int maxAngleJumpSteps = (int)((stepsPerRotation / 360.0f) * (MAX_ANGLE_JUMP_DEGREES * rotationScale * 2));
  memcpy(&incomingcontrol, incomingData, sizeof(incomingcontrol));
  LogDebugFormattedPRIO("RX packet: sender=%d target=%d cmd=%d heartbeat=%d len=%d\n",
                        incomingcontrol.esp_sender,
                        incomingcontrol.esp_target,
                        incomingcontrol.esp_command,
                        incomingcontrol.esp_heartbeat ? 1 : 0,
                        len);

  // Pair M5 when frames are explicitly for Fist-IT, or when M5 sends
  // discovery CONNECT/HEARTBEAT with a different target.
  if (incomingcontrol.esp_sender == M5_ID &&
      (incomingcontrol.esp_target == FIST_ID ||
       incomingcontrol.esp_command == CONNECT ||
       incomingcontrol.esp_command == HEARTBEAT)) {
    if (!M5_paired) {
      esp_now_del_peer(peerInfo.peer_addr);
      memcpy(M5_Remote_Address, mac, 6);
      memcpy(peerInfo.peer_addr, mac, 6);
      peerInfo.channel = 1;
      peerInfo.encrypt = false;
      esp_err_t pairResult = esp_now_add_peer(&peerInfo);
      if (pairResult == ESP_OK || pairResult == ESP_ERR_ESPNOW_EXIST) {
        M5_paired = true;
        LogDebugFormattedPRIO("M5 paired from frame target=%d cmd=%d: %02X:%02X:%02X:%02X:%02X:%02X\n",
                              incomingcontrol.esp_target,
                  incomingcontrol.esp_command,
                              mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]);
      } else {
        LogDebugFormattedPRIO("M5 pair add failed from target=%d cmd=%d frame: %d\n",
                              incomingcontrol.esp_target,
                incomingcontrol.esp_command,
                              pairResult);
      }
    } else {
      memcpy(M5_Remote_Address, mac, 6);
    }
  }

    // ALWAYS ALLOW CONNECTION AND HEARTBEAT COMMANDS - they establish communication
    if (incomingcontrol.esp_command == CONNECT || incomingcontrol.esp_command == HEARTBEAT) {
      LogDebugFormattedPRIO("CONNECT/HEARTBEAT received: sender=%d target=%d\n",
                            incomingcontrol.esp_sender,
                            incomingcontrol.esp_target);
      
      // Send immediate response to establish connection
      outgoingcontrol.esp_command = HEARTBEAT; 
      outgoingcontrol.esp_value = 1.0f;
      outgoingcontrol.esp_target = incomingcontrol.esp_sender;
      outgoingcontrol.esp_sender = FIST_ID;
      outgoingcontrol.esp_connected = true;
      esp_now_send(mac, (uint8_t *) &outgoingcontrol, sizeof(outgoingcontrol));
      LogDebugFormattedPRIO("CONNECT response sent to device %d (early return path)\n", incomingcontrol.esp_sender);
      return; // Don't process as other commands
    }
    
    // ALWAYS SEND ACKNOWLEDGMENTS FOR CONNECTION HEARTBEATS
    if (incomingcontrol.esp_heartbeat) {
      // Send heartbeat response to confirm connection is alive
      outgoingcontrol.esp_heartbeat = true;
      outgoingcontrol.esp_connected = true;
      outgoingcontrol.esp_target = incomingcontrol.esp_sender;
      outgoingcontrol.esp_sender = FIST_ID;
      esp_now_send(mac, (uint8_t *) &outgoingcontrol, sizeof(outgoingcontrol));
      LogDebugFormattedPRIO("Heartbeat response sent to device %d\n", incomingcontrol.esp_sender);
    }


    // Use received esp_speed for motor control (minimal change)
    // For gyro, always update velocity and angle if present
    if (incomingcontrol.esp_sender == FIST_IT_GYRO_ID) {
      LogDebug("Gyro data received from FIST_IT_GYRO_ID");
      OnDataRecvGYRO(mac, incomingData, len);

    }

  // SMART PEER MANAGEMENT: Only add peer if not already paired
  if(incomingcontrol.esp_target == FIST_ID) {
    LogDebugFormattedPRIO("Peer mgmt entry: sender=%d target=%d M5_paired=%d OSSM_paired=%d\n",
                          incomingcontrol.esp_sender,
                          incomingcontrol.esp_target,
                          M5_paired ? 1 : 0,
                          OSSM_paired ? 1 : 0);
    
    bool needToAddPeer = false;
    
    if (incomingcontrol.esp_sender == M5_ID && !M5_paired) {
      // M5 Remote trying to connect for first time
      esp_now_del_peer(peerInfo.peer_addr); // Clear any existing peer
      memcpy(M5_Remote_Address, mac, 6);
      memcpy(peerInfo.peer_addr, mac, 6);
      needToAddPeer = true;
      LogDebugPRIO("Setting up M5 Remote peer");
    }
    else if (incomingcontrol.esp_sender == M5_ID && M5_paired) {
      // M5 is already paired, just switch to its peer address
      memcpy(peerInfo.peer_addr, M5_Remote_Address, 6);
      LogDebugPRIO("Switching to M5 Remote peer");
    }
    else if (incomingcontrol.esp_sender == OSSM_ID) {
      LogDebugPRIO("OSSM packet reached peer management block");
    }
    
    if (needToAddPeer) {
      peerInfo.channel = 1;
      peerInfo.encrypt = false;
      
      esp_err_t result = esp_now_add_peer(&peerInfo);
      if (result == ESP_OK) {
        LogDebugFormattedPRIO("Peer added: %02X:%02X:%02X:%02X:%02X:%02X from ID %d\n", 
                          mac[0], mac[1], mac[2], mac[3], mac[4], mac[5], incomingcontrol.esp_sender);
        
        if (incomingcontrol.esp_sender == M5_ID) {
          M5_paired = true;
          LogDebugPRIO("M5 Remote Connected");
        } 
        else if (incomingcontrol.esp_sender == OSSM_ID) {
          OSSM_paired = true;
          LogDebug("OSSM Stroke Engine Connected!");
        }
      }
      else {
        LogDebugPRIO("Failed to add peer");
      }
    }
    // End peer management block
    
    // SAFETY SYSTEM: Check if movement commands should be processed
    bool allowMovementCommand = true;
    unsigned long currentTime = millis();
    
    if (incomingcontrol.esp_sender == M5_ID) {
      // M5 Remote always has movement priority
      lastM5CommandTime = currentTime;
      activeController = M5_ID;
      LogDebug("SAFETY: M5 Remote active - has priority control");
      
      // Check if M5 is sending a stop command (zero velocity)
      bool isStopCommand = (abs(incomingcontrol.esp_value) < 0.01f);
      if (isStopCommand) {
        // Stop command - clear M5 activity to allow gyro control later
        lastM5CommandTime = 0;
        activeController = 0;
        LogDebug("SAFETY: M5 Remote STOP - clearing priority for gyro connection");
      }
    } 
    
    // Process movement commands only if safety allows
    if (!allowMovementCommand) {
      LogDebug("Movement command blocked by safety system");
      return;
    }

    //HANDLE ACTUAL COMMANDS SENT FROM M5 REMOTE

    // --- Minimal Command Execution ---
    switch(incomingcontrol.esp_command)
    {
      case FSPEED: {
        float m5SpeedRaw = incomingcontrol.esp_value;
        speedValue = m5SpeedRaw / 500.0f * maxSpeedHz;
        // Recalculate SensationValue using lastSensationPercent and sensationScale
        SensationValue = (int)(speedValue * (lastSensationPercent / 100.0f) * sensationScale);
        LogDebug(String("[M5 DEBUG] SPEED: raw=") + String(m5SpeedRaw) + ", mapped speedValue=" + String(speedValue) + ", SensationValue recalculated=" + String(SensationValue) + " (" + String(lastSensationPercent) + "% of speedValue, scale=" + String(sensationScale) + ")");
        break;
      }
      case SPEED: {
        float m5SpeedRaw = incomingcontrol.esp_value;
        speedValue = m5SpeedRaw / 100.0f * maxSpeedHz;
        // Recalculate SensationValue using lastSensationPercent and sensationScale
        SensationValue = (int)(speedValue * (lastSensationPercent / 100.0f) * sensationScale);
        LogDebug(String("[M5 DEBUG] SPEED: raw=") + String(m5SpeedRaw) + ", mapped speedValue=" + String(speedValue) + ", SensationValue recalculated=" + String(SensationValue) + " (" + String(lastSensationPercent) + "% of speedValue, scale=" + String(sensationScale) + ")");
        break;
      }
      case FROTATION: {
        // Only print debug for new angle/speed from gyro above
        EndPosition = (int)((stepsPerRotation/360.0) * (incomingcontrol.esp_value * rotationScale) * 4);
        //MotorStatus = START;
        //currentRotationVelocity = incomingcontrol.esp_speed;
        //speedValue = (int)incomingcontrol.esp_speed;
        break;
      }
      case FPAUSE: {
        F_Pause = incomingcontrol.esp_value;
        break;
      }
      case FSENSATION: {
        LogDebug("FSENSATION command received");
        // SensationValue is a percentage of speedValue, scaled by sensationScale
        float requestedPercent = incomingcontrol.esp_value; // 0-100 from remote
        lastSensationPercent = requestedPercent;
        SensationValue = (int)(speedValue * (requestedPercent / 100.0f) * sensationScale);
        LogDebug(String("[FSENSATION] SensationValue=") + String(SensationValue) + " (" + String(requestedPercent) + "% of speedValue=" + String(speedValue) + ", scale=" + String(sensationScale) + ")");
        break;
      }
      case FANGLE: {
        // FANGLE removed: Only FROTATION is used for gyro control
        break;
      }
      case FROTATION_SCALE: {
        rotationScale = incomingcontrol.esp_value;
        break;
      }
      case FRESPONSIVENESS: {
        responsivenessLevel = (int)incomingcontrol.esp_value;
        break;
      }
      case FVELOCITY: {
        // Only print debug for new angle/speed from gyro above
        currentRotationVelocity = incomingcontrol.esp_value;
        break;
      }
      case ON: {
        MotorStatus=START;
        break;
      }
      case OFF: {
        MotorStatus=STOP;
        break;
      }
      case CONNECT:
      case HEARTBEAT: {
        // ...existing code...
        if (incomingcontrol.esp_sender == M5_ID) {
          outgoingcontrol.esp_target = M5_ID;
          outgoingcontrol.esp_sender = FIST_ID;
          outgoingcontrol.esp_command = HEARTBEAT;
          esp_err_t result = esp_now_send(M5_Remote_Address, (uint8_t *) &outgoingcontrol, sizeof(outgoingcontrol));
          if (result == ESP_OK) {
            M5_paired = true;
          }
        }
        else if (incomingcontrol.esp_sender == FIST_IT_GYRO_ID) {
          outgoingcontrol.esp_target = FIST_IT_GYRO_ID;
          outgoingcontrol.esp_sender = FIST_ID;
          outgoingcontrol.esp_command = HEARTBEAT;
          esp_err_t result = esp_now_send(mac, (uint8_t *) &outgoingcontrol, sizeof(outgoingcontrol));
          if (result == ESP_OK) {
            FIST_IT_RC_paired = true;
          }
        }
        break;
      }
    }
    
    // Send appropriate response directly to the sender (no peer switching needed for responses)
    outgoingcontrol.esp_sender = FIST_ID;
    outgoingcontrol.esp_command = CONNECT;
    outgoingcontrol.esp_value = 99;
    
    if (incomingcontrol.esp_sender == M5_ID) {
      outgoingcontrol.esp_target = M5_ID;
    } else if (incomingcontrol.esp_sender == FIST_IT_GYRO_ID) {
      outgoingcontrol.esp_target = FIST_IT_GYRO_ID;
    }
    
    // Send response directly to sender's MAC address
    esp_err_t sendResult = esp_now_send(mac, (uint8_t *) &outgoingcontrol, sizeof(outgoingcontrol));

  } 
} // Close OnDataRecv function


// Initialize ESP-NOW communication
void setupESPNow() {
    // WiFi setup is now done during splash screen, so skip setupWiFiManager()
    
    // Get the actual WiFi channel and use it for ESP-NOW
    int32_t actualChannel;
    if (WiFi.status() == WL_CONNECTED) {
        actualChannel = WiFi.channel();
        LogDebugFormattedPRIO("WiFi network '%s' is on channel: %d\n", WiFi.SSID().c_str(), actualChannel);
    } else {
        // If not connected to WiFi, use default channel 1
        actualChannel = 1;
        LogDebugPRIO("WiFi not connected, using default channel 1 for ESP-NOW");
    }
    
    actualChannel = 1; // Force channel 1 for compatibility with Fist-IT Motor

    // Set ESP-NOW to use the determined channel
    esp_wifi_set_promiscuous(true);
    esp_wifi_set_channel(actualChannel, WIFI_SECOND_CHAN_NONE);
    esp_wifi_set_promiscuous(false);
    
    LogDebugFormattedPRIO("ESP-NOW channel set to: %d (matches WiFi)\n", actualChannel);
    
    // Print our own MAC address for reference
    //Serial.print("Fist-IT Gyro Controller MAC: ");
    LogDebugPRIO(WiFi.macAddress());
    
    // Initialize ESP-NOW
    //Serial.print("Initializing ESP-NOW... ");
    if (esp_now_init() != ESP_OK) {
        LogDebug("Error initializing ESP-NOW");
        return;
    }
    LogDebug("Success!");
    
    // Register callback functions
      esp_now_register_send_cb(OnDataSent);
  
      // Initialize single dynamic peer info structure
      memset(&peerInfo, 0, sizeof(peerInfo));
      peerInfo.channel = 1;  
      peerInfo.encrypt = false;
    
      // Register for a callback function that will be called when data is received
      esp_now_register_recv_cb(OnDataRecv);
    
    // Set up broadcast peer for initial discovery
    memset(&peerInfo, 0, sizeof(peerInfo));
    uint8_t broadcastAddress[] = {0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF};
    memcpy(peerInfo.peer_addr, broadcastAddress, 6);
    peerInfo.channel = 1;  
    peerInfo.encrypt = false;
    
    // Add broadcast peer
    esp_err_t peerResult = esp_now_add_peer(&peerInfo);
    if (peerResult != ESP_OK) {
        LogDebugFormatted("Failed to add broadcast peer, error: %d\n", peerResult);
        return;
    }
    
    LogDebugPRIO("ESP-NOW initialized with broadcast peer");
    LogDebug("Sending discovery messages to find devices...");
    
}

void moveFromM5Remote(){
  // Function to handle movement commands from M5 Remote
  // Currently integrated into OnDataRecv for minimal changes
  
  if (MotorStatus == START) {
    // M5 remote: use speedValue and SensationValue for speed/accel
    //LogDebugPRIO(String("[M5 DEBUG] MotorStatus=START, speedValue=") + String(speedValue) + ", SensationValue=" + String(SensationValue));
    int multiplier = 1;
    int speedHz = speedValue * multiplier;
    int accel = SensationValue * multiplier;
    if (speedHz < minSpeedHz) speedHz = minSpeedHz;
    if (speedHz > maxSpeedHz) speedHz = maxSpeedHz;
    if (accel < 100) accel = 100;
    //EndPosition=EndPosition*4;
    //if (accel > 40000) accel = 40000;
    //LogDebugPRIO(String("[M5 DEBUG] Calculated speedHz: ") + String(speedHz) + ", accel: " + String(accel));
    //LogDebugPRIO(String("[M5 DEBUG] CurrentPosition: ") + String(stepper->getCurrentPosition()));
    stepper->setSpeedInTicks(((multiplier*10000)+1)-speedHz);
    stepper->setAcceleration(accel);
    // Oscillate between EndPosition and StartPosition with blocking pause after each move
    if (stepper->targetPos() == stepper->getCurrentPosition()) {
      if (stepper->getCurrentPosition() == StartPosition) {
        stepper->moveTo(EndPosition);
        LogDebugPRIO("[M5 DEBUG] Moving to EndPosition: " + String(EndPosition) + " Pause time = " + String(F_Pause));
        delay(F_Pause * 100);
      } else if (stepper->getCurrentPosition() == EndPosition) {
        stepper->moveTo(StartPosition);
        LogDebugPRIO("[M5 DEBUG] Moving to StartPosition: " + String(StartPosition) + " Pause time = " + String(F_Pause));
        delay(F_Pause * 100);
      } else if (stepper->getCurrentPosition() < EndPosition) {
        stepper->moveTo(EndPosition);
        LogDebugPRIO("[M5 DEBUG] Moving to EndPosition: " + String(EndPosition) + " Pause time = " + String(F_Pause));
        delay(F_Pause * 100);
      } else {
        stepper->moveTo(StartPosition);
        LogDebugPRIO("[M5 DEBUG] Moving to StartPosition: " + String(StartPosition) + " Pause time = " + String(F_Pause));
        delay(F_Pause * 100);
      }
    }
  }
}


void moveFromGyro(){
  if (MotorStatus != ANGLE_MODE) return;
  int speedHz = abs(speedValue);
  switch(speedHz){
    case 0 ... 15:
      speedHz = 0;
      break;
    case 16 ... 100:
      speedHz = 100;
      break;
    case 101 ... 200:
      speedHz = 200;
      break;
    case 201 ... 1000:
      speedHz = (map(speedHz,201,1000,201,1000)/50)*50;
      break;
    case 1001 ... 5000:
      speedHz = (map(speedHz,1001,5000,1001,5000)/250)*250;
      break;
    case 5001 ... 100000:
      speedHz = (map(speedHz,5001,100000,5001,100000)/1000)*1000;
      break;
    default:
      speedHz = 10000;
      break;
  }
  const int threshold = 5; // Minimum distance to move (tune as needed)
  const int maxStepPerCall = 50; // Maximum steps per call (tune as needed)
  int distanceToGo = EndPosition - stepper->getCurrentPosition();
  stepper->setSpeedInHz(speedHz);
  stepper->setAcceleration(speedHz*1.5);
  if(abs(distanceToGo)<threshold+maxStepPerCall+1) {
    stepper->forceStop();
    //stepper->setCurrentPosition(0);
    LogDebugPRIO("[GYRO] ANGLE_MODE: distanceToGo < 1, motor stopped. Current position: " + String(stepper->getCurrentPosition()) + ", EndPosition: " + String(EndPosition) + ", distanceToGo: " + String(distanceToGo) );
    return;
  }
  if (abs(distanceToGo) > threshold) {
    int stepChunk = constrain(abs(distanceToGo), 1, maxStepPerCall);
    // Clamp stepChunk to not overshoot
    if (stepChunk > abs(distanceToGo)) {
      stepChunk = abs(distanceToGo);
    }
    int direction = (distanceToGo > 0) ? 1 : -1;
    stepper->move(direction * stepChunk);
//    LogDebugPRIO("[GYRO] Moving: stepChunk=" + String(stepChunk) + ", direction=" + String(direction) + ", EndPosition=" + String(EndPosition) + ", CurrentPosition=" + String(stepper->getCurrentPosition()));
    // After move, check if we've reached or passed EndPosition
    int newPos = stepper->getCurrentPosition() + direction * stepChunk;
    if ((direction > 0 && newPos >= EndPosition) || (direction < 0 && newPos <= EndPosition)) {
      stepper->forceStop();
      stepper->setCurrentPosition(EndPosition);
      LogDebugPRIO("[GYRO] Clamped to EndPosition: " + String(EndPosition));
    }
  } else {
    // Already at or very close to EndPosition
    stepper->forceStop();
    stepper->setCurrentPosition(EndPosition);
//    LogDebugPRIO("[GYRO] At EndPosition: " + String(EndPosition));
  }
  LogDebugPRIO("Speedvalue: " + String(abs(speedHz)) +", EndPosition: " + String(EndPosition) + ", CurrentPosition: " + String(stepper->getCurrentPosition()) + ", DistanceToGo: " + String((distanceToGo))  );
  delay(20);
}
